## Schéma Prisma
```prisma
model Order {
  id        String      @id @default(cuid())
  name      String
  phone     String
  address   String
  mode      String      // "livraison" | "surplace"
  notes     String?
  total     Float
  status    String      @default("pending")
  items     OrderItem[]
  createdAt DateTime    @default(now())
}

model OrderItem {
  id       String  @id @default(cuid())
  orderId  String
  order    Order   @relation(fields: [orderId], references: [id])
  name     String
  price    Float
  quantity Int
  options  String? // JSON string pour les options tacos
}
```
